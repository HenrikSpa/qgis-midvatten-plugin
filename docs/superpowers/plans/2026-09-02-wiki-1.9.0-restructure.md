**Goal:** Rebuild the Midvatten GitHub wiki so it documents plugin 1.9.0 exactly, with a regenerable screenshot pipeline (QGIS 4.2.2 / Qt6 in Docker), a tutorial dataset covering the 1.11.1 schema, and check scripts that keep labels, figures and links honest.

```python
#!/usr/bin/env python3
"""Wiki consistency gate: links, forbidden tokens, figure manifest, feature labels.

Usage: python3 tools/check_wiki.py [--root DIR] [--labels tools/generated/actions.txt]
       [--manifest tools/screenshots/manifest.toml]
Exit code 1 if any problem is found. Pure stdlib; runs on the host.
"""
from __future__ import annotations

import argparse
import re
import sys
import tomllib
from pathlib import Path

HOWTO_PAGES = {"Import-data", "Export-data", "Edit-data", "Plots", "Reports",
               "Database-management", "Utilities"}
EXEMPT_PREFIXES = ("Legacy", "Release-notes", "Tutorial", "8b.", "2b.", "database-historical")
FORBIDDEN = [r"\bv2\.0\b", r"Version 2\.0", r"Figure needed", r"jkall/qgis-midvatten-plugin/wiki",
             r"\bF7\b", r"\bF8\b", r"\bF9\b", r"\bF1[012]\b", r"Upgrade database", r"Calculate statistics"]
WIKILINK = re.compile(r"\[\[([^\]|]+)(?:\|([^\]]+))?\]\]")
MDLINK = re.compile(r"!?\[[^\]]*\]\(([^)\s]+)\)")


def find_pages(root: Path) -> list[Path]:
    return sorted(p for p in root.rglob("*.md") if ".git" not in p.parts)


def _is_exempt(page: Path) -> bool:
    return page.name.startswith(EXEMPT_PREFIXES)


def _page_exists(root: Path, target: str) -> bool:
    name = target.split("#", 1)[0].strip()
    if not name or name.startswith(("http://", "https://", "mailto:")):
        return True
    return (root / f"{name}.md").exists() or (root / name).exists()


def check_links(root: Path, pages: list[Path]) -> list[str]:
    problems = []
    for page in pages:
        text = page.read_text(encoding="utf-8")
        rel = page.relative_to(root)
        for m in WIKILINK.finditer(text):
            target = m.group(2) or m.group(1)
            if not _page_exists(root, target):
                problems.append(f"{rel}: wiki link to missing page '{target}'")
        for m in MDLINK.finditer(text):
            target = m.group(1)
            if target.startswith(("http://", "https://", "mailto:", "#")):
                continue
            if not _page_exists(root, target):
                problems.append(f"{rel}: link/image to missing target '{target}'")
    return problems


def check_tokens(pages: list[Path]) -> list[str]:
    problems = []
    for page in pages:
        if _is_exempt(page):
            continue
        for lineno, line in enumerate(page.read_text(encoding="utf-8").splitlines(), 1):
            for pattern in FORBIDDEN:
                if re.search(pattern, line):
                    problems.append(f"{page.name}:{lineno}: forbidden token /{pattern}/")
    return problems


def check_figures(root: Path, pages: list[Path], manifest_path: Path) -> list[str]:
    manifest = tomllib.loads(manifest_path.read_text(encoding="utf-8"))
    referenced: dict[str, set[str]] = {}
    for page in pages:
        for m in re.finditer(r"!\[[^\]]*\]\(images/1\.9\.0/([^)]+)\.png\)", page.read_text(encoding="utf-8")):
            referenced.setdefault(m.group(1), set()).add(page.stem)
    problems = []
    for name in manifest:
        if name not in referenced:
            problems.append(f"manifest figure '{name}' is not referenced by any page")
        elif not (root / "images/1.9.0" / f"{name}.png").exists():
            problems.append(f"manifest figure '{name}' has no PNG under images/1.9.0/")
    for name, pages_using in referenced.items():
        if name not in manifest:
            problems.append(f"figure '{name}' used by {sorted(pages_using)} is missing from manifest")
    return problems


def load_labels(actions_txt: Path) -> set[str]:
    labels = set()
    for line in actions_txt.read_text(encoding="utf-8").splitlines():
        cells = [c.strip() for c in line.strip().strip("|").split("|")]
        if len(cells) >= 2 and cells[0] not in ("Submenu", "---") and not cells[0].startswith("-"):
            labels.add(cells[1])
    return labels


def check_labels(pages: list[Path], actions_txt: Path) -> list[str]:
    labels = load_labels(actions_txt) | {"Midvatten Settings", "About", "Reset settings", "Load default db-layers to qgis"}
    problems = []
    for page in pages:
        if page.stem not in HOWTO_PAGES:
            continue
        for line in page.read_text(encoding="utf-8").splitlines():
            if line.startswith("### "):
                heading = line[4:].strip()
                if heading not in labels:
                    problems.append(f"{page.name}: heading '{heading}' is not a menu label")
    return problems


def main(argv: list[str] | None = None) -> int:
    ap = argparse.ArgumentParser(description=__doc__)
    ap.add_argument("--root", default=".", type=Path)
    ap.add_argument("--labels", default="tools/generated/actions.txt", type=Path)
    ap.add_argument("--manifest", default="tools/screenshots/manifest.toml", type=Path)
    args = ap.parse_args(argv)
    root = args.root.resolve()
    pages = find_pages(root)
    problems = check_links(root, pages) + check_tokens(pages)
    if (root / args.manifest).exists():
        problems += check_figures(root, pages, root / args.manifest)
    if (root / args.labels).exists():
        problems += check_labels(pages, root / args.labels)
    for p in problems:
        print(p)
    print(f"{len(problems)} problem(s)")
    return 1 if problems else 0


if __name__ == "__main__":
    sys.exit(main())
```

```python
#!/usr/bin/env python3
"""Host CLI for the wiki screenshot pipeline (Docker + QGIS 4.2.2/Qt6 + Xvfb).

  shoot.py build                       build image midvatten-docs:4.2.2
  shoot.py exec [--gui] -- CMD...      run CMD inside the container (cwd /wiki)
  shoot.py shoot --all | --figures a,b [--dpi 1] [--out images/1.9.0] [--db PATH]
  shoot.py gen-actions                 write tools/generated/actions.txt from the plugin manifest
"""
from __future__ import annotations

import argparse
import json
import os
import subprocess
import sys
from pathlib import Path

WIKI = Path(__file__).resolve().parents[2]
IMAGE = "midvatten-docs:4.2.2"
DEFAULT_PLUGIN = Path(os.environ.get("MIDVATTEN_PLUGIN_DIR", "/home/hsai1/dev/midv/midvatten"))
XVFB = ["xvfb-run", "-a", "-s", "-screen 0 1600x1000x24"]
RUNNER = "/wiki/tools/screenshots/runner.py"
DEFAULT_DB = "/wiki/tutorial_data/build/tutorial.sqlite"
DEFAULT_PROJECT = "/wiki/tutorial_data/build/midvatten_tutorial.qgz"


def container_cmd(args: list[str], gui: bool, plugin_dir: Path = DEFAULT_PLUGIN, env: dict[str, str] | None = None) -> list[str]:
    cmd = ["docker", "run", "--rm", "--user", f"{os.getuid()}:{os.getgid()}", "-e", "HOME=/tmp",
           "-v", f"{WIKI}:/wiki", "-v", f"{plugin_dir}:/plugin:ro", "-w", "/wiki"]
    for key, value in (env or {}).items():
        cmd += ["-e", f"{key}={value}"]
    cmd.append(IMAGE)
    return cmd + (XVFB + args if gui else args)


def run(cmd: list[str]) -> None:
    print("+", " ".join(cmd), file=sys.stderr)
    subprocess.run(cmd, check=True)


def cmd_build(_: argparse.Namespace) -> int:
    run(["docker", "build", "-t", IMAGE, str(WIKI / "tools/screenshots")])
    return 0


def cmd_exec(ns: argparse.Namespace) -> int:
    run(container_cmd(ns.cmd, gui=ns.gui, plugin_dir=ns.plugin))
    return 0
```

```python
"""Runs inside QGIS via `qgis --code runner.py --py-args ... --`.

Loads the Midvatten plugin, opens the tutorial database, then executes the
scene modules that produce the requested figures. Never imported on the host.
"""
from __future__ import annotations

import argparse
import importlib
import json
import sys
import tomllib
import traceback
from pathlib import Path

import qgis.utils
from qgis.core import QgsApplication, QgsProject
from qgis.PyQt.QtCore import QEventLoop, QTimer, Qt
from qgis.PyQt.QtWidgets import QApplication, QDialog, QWidget

HERE = Path("/wiki/tools/screenshots")
RESULT_FILE = HERE / ".last_run.json"


class Context:
    def __init__(self, plugin, iface, out_dir: Path):
        self.plugin = plugin
        self.iface = iface
        self.out_dir = out_dir
        self.out_dir.mkdir(parents=True, exist_ok=True)
        self.produced: list[str] = []

    # --- time ---------------------------------------------------------
    def wait(self, ms: int) -> None:
        loop = QEventLoop()
        QTimer.singleShot(ms, loop.quit)
        loop.exec()

    # --- layers -------------------------------------------------------
    def layer(self, name: str):
        layers = QgsProject.instance().mapLayersByName(name)
        if not layers:
            raise LookupError(f"layer {name!r} not loaded; load default layers first")
        return layers[0]

    def select(self, layer_name: str, expression: str) -> int:
        lyr = self.layer(layer_name)
        lyr.selectByExpression(expression)
        return lyr.selectedFeatureCount()

    def activate(self, layer_name: str):
        lyr = self.layer(layer_name)
        self.iface.setActiveLayer(lyr)
        return lyr

    # --- plugin actions -------------------------------------------------
    def open_action(self, action_id: str) -> QWidget:
        spec = next(s for s in self.plugin._actions_manifest if s.id == action_id)
        before = {id(w) for w in QApplication.topLevelWidgets()}
        self.plugin._dispatch(spec)
        self.wait(1500)
        tool = self.plugin._open_tools.get(action_id)
        if tool is not None and isinstance(tool, QWidget):
            return tool
        new = [w for w in QApplication.topLevelWidgets()
               if id(w) not in before and w.isVisible() and w is not self.iface.mainWindow()]
        if not new:
            raise RuntimeError(f"action {action_id} opened no visible window")
        return max(new, key=lambda w: w.width() * w.height())

    def close_tools(self) -> None:
        for w in list(QApplication.topLevelWidgets()):
            if w is self.iface.mainWindow() or not w.isVisible():
                continue
            if isinstance(w, QDialog):
                w.reject()
            else:
                w.close()
        self.wait(300)

    # --- capture --------------------------------------------------------
    def grab(self, widget: QWidget, figure: str) -> Path:
        self.wait(400)
        widget.raise_()
        QApplication.processEvents()
        path = self.out_dir / f"{figure}.png"
        if not widget.grab().save(str(path)):
            raise RuntimeError(f"could not save {path}")
        self.produced.append(figure)
        return path

    def grab_modal(self, figure: str, delay_ms: int = 800, close: str = "reject") -> None:
        """Schedule a grab of the next modal dialog (for tools that call exec())."""
        def _capture():
            dlg = QApplication.activeModalWidget()
            if dlg is None:
                self.produced.append(f"error:{figure}: no modal dialog appeared")
                return
            self.grab(dlg, figure)
            getattr(dlg, close)()
        QTimer.singleShot(delay_ms, _capture)

    def find_child(self, widget: QWidget, object_name: str) -> QWidget:
        child = widget.findChild(QWidget, object_name)
        if child is None:
            names = sorted({c.objectName() for c in widget.findChildren(QWidget) if c.objectName()})
            raise LookupError(f"no child {object_name!r}; available: {names}")
        return child

    def main_window(self):
        return self.iface.mainWindow()


def parse_args() -> argparse.Namespace:
    ap = argparse.ArgumentParser()
    ap.add_argument("--db", default="")
    ap.add_argument("--project", default="")
    ap.add_argument("--out", default="/wiki/images/1.9.0")
    ap.add_argument("--figures", default="")
    ap.add_argument("--gen-actions", default="")
    argv = [a for a in sys.argv[1:] if a != "--"]
    return ap.parse_args(argv)


def start_plugin():
    qgis.utils.loadPlugin("midvatten")
    qgis.utils.startPlugin("midvatten")
    return qgis.utils.plugins["midvatten"]


def open_database(plugin, dbpath: str) -> None:
    settings = json.dumps({"spatialite": {"dbpath": dbpath}})
    QgsProject.instance().writeEntry("Midvatten", "database", settings)
    plugin.ms.load_settings()


def write_actions(plugin, path: str) -> None:
    submenus = {"import": "Import data to database", "export": "Export data from database",
                "edit": "Edit data in database", "plot": "Plots", "report": "Reports",
                "db": "Database management", "utils": "Utilities"}
    lines = ["| Submenu | Label | id | Toolbar | Requires |", "|---|---|---|---|---|"]
    for s in plugin._actions_manifest:
        req = []
        if s.needs_db:
            req.append("db")
        if s.critical_layers:
            req.append("layers: " + ", ".join(s.critical_layers))
        if s.needs_selection:
            req.append("selection")
        if s.needs_active_layer:
            req.append(f"active layer: {s.needs_active_layer}")
        lines.append(f"| {submenus[s.menu]} | {s.label} | {s.id} | {'yes' if s.toolbar else 'no'} | {'; '.join(req)} |")
    Path(path).write_text("\n".join(lines) + "\n", encoding="utf-8")


def run_scenes(ctx: Context, figures: list[str]) -> dict[str, str]:
    manifest = tomllib.loads((HERE / "manifest.toml").read_text(encoding="utf-8"))
    wanted = list(manifest) if figures == ["ALL"] else figures
    unknown = [f for f in wanted if f not in manifest]
    if unknown:
        raise SystemExit(f"figures not in manifest: {unknown}")
    results: dict[str, str] = {}
    sys.path.insert(0, str(HERE))
    for module_name in dict.fromkeys(manifest[f]["scene"] for f in wanted):
        module = importlib.import_module(f"scenes.{module_name}")
        expected = [f for f in module.FIGURES if f in wanted]
        try:
            module.scene(ctx)
            ctx.wait(1500)  # let grab_modal callbacks fire
        except Exception:
            tb = traceback.format_exc().strip().splitlines()[-1]
            for f in expected:
                results[f] = f"error: {tb}"
            ctx.close_tools()
            continue
        for f in expected:
            results[f] = "ok" if f in ctx.produced else "error: scene finished without producing it"
        for entry in ctx.produced:
            if entry.startswith("error:"):
                _, name, msg = entry.split(":", 2)
                results[name] = f"error:{msg}"
        ctx.close_tools()
    return results


def main() -> None:
    ns = parse_args()
    results: dict[str, str] = {}
    try:
        plugin = start_plugin()
        if ns.gen_actions:
            write_actions(plugin, ns.gen_actions)
            results["gen-actions"] = "ok"
        else:
            if ns.db:
                open_database(plugin, ns.db)
            if ns.project and Path(ns.project).exists():
                QgsProject.instance().read(ns.project)
                open_database(plugin, ns.db)  # project read resets the entry
            mw = qgis.utils.iface.mainWindow()
            mw.resize(1600, 1000)
            mw.showMaximized()
            ctx = Context(plugin, qgis.utils.iface, Path(ns.out))
            ctx.wait(2000)
            results = run_scenes(ctx, ns.figures.split(",") if ns.figures else [])
    except Exception:
        results["runner"] = "error: " + traceback.format_exc().strip().splitlines()[-1]
        traceback.print_exc()
    RESULT_FILE.write_text(json.dumps(results, indent=2))
    QTimer.singleShot(500, QgsApplication.instance().quit)


QTimer.singleShot(3000, main)
```

```python
#!/usr/bin/env python3
"""Generate tools/generated/schema.txt from tutorial.sqlite and definitions/create_db.sql.

Usage: python3 tools/gen_schema.py --db tutorial_data/build/tutorial.sqlite \
         --sql /home/hsai1/dev/midv/midvatten/definitions/create_db.sql --out tools/generated/schema.txt
Column descriptions come from the `--` comments in create_db.sql; structure from PRAGMAs.
"""
from __future__ import annotations

import argparse
import re
import sqlite3
import sys
from pathlib import Path

TABLE_RE = re.compile(r"CREATE TABLE (\w+)\s*(?:/\*(.*?)\*/)?\s*\(", re.S)
COL_RE = re.compile(r"^(?:SPATIALITE |POSTGIS )?,?\s*(\w+)\s+[\w ]+?.*?--\s*(.*)$")


def parse_column_comments(sql_text: str) -> dict[str, dict[str, str]]:
    info: dict[str, dict[str, str]] = {}
    current = None
    for line in sql_text.splitlines():
        m = TABLE_RE.search(line)
        if m:
            current = m.group(1)
            info[current] = {"__table__": (m.group(2) or "").strip()}
            continue
        if current is None:
            continue
        if line.strip().startswith(")"):
            current = None
            continue
        stripped = line.strip()
        if stripped.startswith(("SPATIALITE", "POSTGIS", ",")) or re.match(r"^\w+ ", stripped):
            body = re.sub(r"^(SPATIALITE|POSTGIS)\s+", "", stripped).lstrip(", ")
            if body.upper().startswith(("UNIQUE", "FOREIGN", "PRIMARY", "CHECK")):
                continue
            name = body.split()[0]
            comment = body.split("--", 1)[1].strip() if "--" in body else ""
            if comment:
                info[current][name] = comment
    return info


def describe_table(conn: sqlite3.Connection, name: str) -> list[tuple[str, str, str]]:
    fks = {}
    for (_id, _seq, table, frm, to, _u, on_delete, _m) in conn.execute(f"PRAGMA foreign_key_list('{name}')"):
        extra = f" (ON DELETE {on_delete})" if on_delete and on_delete != "NO ACTION" else ""
        fks[frm] = f"FK → {table}.{to}{extra}"
    rows = []
    for (_cid, col, ctype, notnull, _default, pk) in conn.execute(f"PRAGMA table_info('{name}')"):
        flags = []
        if pk:
            flags.append("PK")
        if notnull:
            flags.append("NOT NULL")
        if col in fks:
            flags.append(fks[col])
        rows.append((col, ctype, ", ".join(flags)))
    return rows


def unique_indexes(conn: sqlite3.Connection, name: str) -> list[str]:
    out = []
    for (_seq, idx, unique, origin, _p) in conn.execute(f"PRAGMA index_list('{name}')"):
        if unique and origin != "pk":
            sql = conn.execute("SELECT sql FROM sqlite_master WHERE name=?", (idx,)).fetchone()[0] or ""
            cols = sql.split("(", 1)[1].rstrip(")") if "(" in sql else idx
            out.append(cols)
    return out


def render(conn: sqlite3.Connection, comments: dict[str, dict[str, str]]) -> str:
    lines = []
    tables = [r[0] for r in conn.execute(
        "SELECT name FROM sqlite_master WHERE type='table' AND name NOT LIKE 'sqlite_%' "
        "AND name NOT LIKE 'geometry_columns%' AND name NOT LIKE 'spatial_ref_sys%' AND name NOT LIKE 'idx_%' "
        "AND name NOT IN ('SpatialIndex','ElementaryGeometries','KNN','KNN2','geom_cols_ref_sys','sql_statements_log',"
        "'spatialite_history','views_geometry_columns','virts_geometry_columns','data_licenses','layer_styles') "
        "AND name NOT LIKE '%_auth' AND name NOT LIKE '%_statistics' AND name NOT LIKE '%_field_infos' "
        "AND name NOT LIKE 'vector_%' AND name NOT LIKE 'raster_%' AND name NOT LIKE 'topology%' AND name NOT LIKE 'networks%' "
        "AND name NOT LIKE 'stored_%' AND name NOT LIKE 'iso_metadata%' AND name NOT LIKE 'wms_%' AND name NOT LIKE 'SE_%' AND name NOT LIKE 'rl2map_%' "
        "ORDER BY name")]
    for t in tables:
        desc = comments.get(t, {})
        lines += [f"## {t}", "", desc.get("__table__", ""), "", "| Column | Type | Constraints | Description |", "|---|---|---|---|"]
        for col, ctype, flags in describe_table(conn, t):
            lines.append(f"| {col} | {ctype} | {flags} | {desc.get(col, '')} |")
        for cols in unique_indexes(conn, t):
            lines.append(f"\nUnique: `{cols}`")
        lines.append("")
    lines += ["## Views", ""]
    for (name,) in conn.execute("SELECT name FROM sqlite_master WHERE type='view' ORDER BY name"):
        lines.append(f"- `{name}`")
    lines += ["", "## Triggers", ""]
    for (name, tbl) in conn.execute("SELECT name, tbl_name FROM sqlite_master WHERE type='trigger' ORDER BY name"):
        lines.append(f"- `{name}` on `{tbl}`")
    return "\n".join(lines) + "\n"


def main(argv: list[str] | None = None) -> int:
    ap = argparse.ArgumentParser(description=__doc__)
    ap.add_argument("--db", required=True, type=Path)
    ap.add_argument("--sql", required=True, type=Path)
    ap.add_argument("--out", required=True, type=Path)
    ns = ap.parse_args(argv)
    conn = sqlite3.connect(f"file:{ns.db}?mode=ro", uri=True)
    ns.out.parent.mkdir(parents=True, exist_ok=True)
    ns.out.write_text(render(conn, parse_column_comments(ns.sql.read_text(encoding="utf-8"))), encoding="utf-8")
    print(f"wrote {ns.out}")
    return 0


if __name__ == "__main__":
    sys.exit(main())
```

```python
"""Writers for the native import formats used by the Midvatten tutorial dataset.

Layouts mirror what the plugin parsers accept (tools/import_logger/parsers.py,
tools/import_interlab4.py, tools/import_fieldlogger.py). Pure stdlib.
"""
from __future__ import annotations

import csv
from datetime import datetime
from pathlib import Path
from typing import Callable

LoggerRow = tuple[datetime, float | None, float | None]


def _mon(location: str, serial: str, rows: list[LoggerRow], channels: list[str]) -> str:
    lines = ["[Logger settings]", f"  Serial number           =..00-{serial}  214.", f"  Location                ={location}",
             f"  Number of channels      ={len(channels)}"]
    for i, name in enumerate(channels, 1):
        lines += [f"[Channel {i}]", f"  Identification          ={name}"]
    lines += ["[Data]", str(len(rows))]
    for dt, head, temp in rows:
        fields = "".join(f"{'' if v is None else f'{v:.3f}':>12}" for v in (head, temp)[: len(channels)])
        lines.append(f"{dt:%Y/%m/%d %H:%M:%S}.0{fields}")
    lines.append("END OF DATA FILE OF DATALOGGER FOR WINDOWS")
    return "\n".join(lines) + "\n"


def write_diveroffice_mon(path: Path, location: str, serial: str, rows: list[LoggerRow], channel: str = "WATER HEAD (WC)") -> None:
    Path(path).write_text(_mon(location, serial, rows, [channel, "TEMPERATURE"]), encoding="utf-8")


def write_diveroffice_baro_mon(path: Path, location: str, serial: str, rows: list[LoggerRow]) -> None:
    Path(path).write_text(_mon(location, serial, rows, ["PRESSURE", "TEMPERATURE"]), encoding="utf-8")


def write_levelogger_csv(path: Path, location: str, serial: str, rows: list[LoggerRow]) -> None:
    lines = [f"Serial_number: {serial}", f"Location: {location}", "LEVEL", "UNIT: cm", "TEMPERATURE",
             "Date;Time;ms;LEVEL;TEMPERATURE"]
    lines += [f"{dt:%Y-%m-%d};{dt:%H:%M:%S};0;{'' if h is None else h};{'' if t is None else t}" for dt, h, t in rows]
    Path(path).write_text("\n".join(lines) + "\n", encoding="utf-8")


def write_hobo_csv(path: Path, label: str, serial: str, rows: list[LoggerRow], tz: str = "GMT+01:00") -> None:
    lines = ['"Plot Title: tutorial"',
             f'"#","Date Time, {tz}","Temp, °C (LGR S/N: {serial}, SEN S/N: {serial}, LBL: {label})"']
    for i, (dt, _head, temp) in enumerate(rows, 1):
        hour = dt.hour % 12 or 12
        ampm = "fm" if dt.hour < 12 else "em"
        lines.append(f'{i},"{dt:%m/%d/%y} {hour:02d}:{dt:%M:%S} {ampm}",{temp}')
    Path(path).write_text("\n".join(lines) + "\n", encoding="utf-8")


_PROVADM = ("Lablittera;Namn;Adress;Postnr;Ort;Kommunkod;Projekt;Laboratorium;Provtyp;Provtagare;Registertyp;"
            "ProvplatsID;Provplatsnamn;Specifik provplats;Provtagningsorsak;Provtyp;Provtypspecifikation;Bedömning;"
            "Kemisk bedömning;Mikrobiologisk bedömning;Kommentar;År;Provtagningsdatum;Provtagningstid;Inlämningsdatum;Inlämningstid;")
_PROVDAT = ("Lablittera;Metodbeteckning;Parameter;Mätvärdetext;Mätvärdetal;Mätvärdetalanm;Enhet;Rapporteringsgräns;"
            "Detektionsgräns;Mätosäkerhet;Mätvärdespår;Parameterbedömning;Kommentar;")


def write_interlab4(path: Path, samples: list[dict], results: list[dict]) -> None:
    lines = ["#Interlab", "#Version=4.0", "#Tecken=UTF-16", "#Textavgränsare=Nej", "#Decimaltecken=,", "#Provadm", _PROVADM]
    for s in samples:
        lines.append(";".join([s["lablittera"], "Midvatten AB", "", "", "", "", "Tutorial", "Demo-Laboratoriet", "GV",
                               s.get("provtagare", ""), "", s["provplatsid"], s["provplatsnamn"], s.get("specifik_provplats", ""),
                               "Kontroll", "Grundvatten", "", "", "", "", "", s["datum"][:4], s["datum"], s["tid"], s["datum"], s["tid"]]) + ";")
    lines += ["#Provdat", _PROVDAT]
    for r in results:
        lines.append(";".join([r["lablittera"], r["metod"], r["parameter"], "", r["varde"], r.get("anm", ""), r["enhet"],
                               "", "", "", "", "", ""]) + ";")
    lines.append("#Slut")
    Path(path).write_text("\n".join(lines) + "\n", encoding="utf-16")


def write_fieldlogger(path: Path, observations: list[dict]) -> None:
    lines = ["Location;Date;Time;Type;Value"]
    for o in observations:
        dt = o["date_time"]
        lines.append(f"{o['sublocation']};{dt:%Y-%m-%d};{dt:%H:%M};{o['parametername']};{o['value']}")
    Path(path).write_text("\n".join(lines) + "\n", encoding="utf-8")


RENAMES = {"obs_points_4326.csv": "obs_points.csv", "obs_lines_4326.csv": "obs_lines.csv",
           "w_levels_v110.csv": "w_levels.csv", "w_flow_v110.csv": "w_flow.csv", "w_qual_field_v110.csv": "w_qual_field.csv",
           "w_qual_lab_v110.csv": "w_qual_lab.csv", "seismic_data_v102.csv": "seismic_data.csv", "vlf_data_v102.csv": "vlf_data.csv",
           "stratigraphy.csv": "stratigraphy.csv", "meteo.csv": "meteo.csv"}


def _transform_wkt(wkt: str, transform: Callable[[float, float], tuple[float, float]]) -> str:
    kind, coords = wkt.split("(", 1)
    pairs = [c.split() for c in coords.rstrip(")").split(",")]
    moved = [transform(float(x), float(y)) for x, y in pairs]
    return f"{kind.strip()} ({', '.join(f'{e} {n}' for e, n in moved)})"


def align_csvs(src_dir: Path, out_dir: Path, transform: Callable[[float, float], tuple[float, float]]) -> dict[str, int]:
    out_dir.mkdir(parents=True, exist_ok=True)
    counts: dict[str, int] = {}
    for src_name, dst_name in RENAMES.items():
        src = Path(src_dir) / src_name
        if not src.exists():
            continue
        text = src.read_text(encoding="utf-8").splitlines()
        delimiter = ";" if text[0].count(";") > text[0].count(",") else ","
        reader = list(csv.reader(text, delimiter=delimiter))
        header, body = reader[0], [r for r in reader[1:] if r]
        if dst_name == "obs_points.csv":
            ei, ni = header.index("east"), header.index("north")
            for row in body:
                east, north = transform(float(row[ei]), float(row[ni]))
                row[ei], row[ni] = str(east), str(north)
        if dst_name == "obs_lines.csv":
            wi = header.index("WKT")
            header[wi] = "geometry"
            for row in body:
                row[wi] = _transform_wkt(row[wi], transform)
        with (Path(out_dir) / dst_name).open("w", newline="", encoding="utf-8") as fh:
            writer = csv.writer(fh, delimiter=delimiter, quoting=csv.QUOTE_MINIMAL)
            writer.writerow(header)
            writer.writerows(body)
        counts[dst_name] = len(body)
    return counts
```

```python
def add_dem(dbpath: Path, out: Path) -> None:

    import numpy as np
    conn = sqlite3.connect(dbpath)
    (xmin, ymin, xmax, ymax) = conn.execute("SELECT MIN(east)-500, MIN(north)-500, MAX(east)+500, MAX(north)+500 FROM obs_points").fetchone()
    conn.close()
    nx, ny = 200, 200
    xs = np.linspace(xmin, xmax, nx); ys = np.linspace(ymax, ymin, ny)
    grid = 40.0 + 6.0 * np.sin((xs[None, :] - xmin) / 400.0) + 4.0 * np.cos((ys[:, None] - ymin) / 300.0)
    (out / "dem").mkdir(exist_ok=True)
    ds = gdal.GetDriverByName("GTiff").Create(str(out / "dem/dem_3006.tif"), nx, ny, 1, gdal.GDT_Float32)

    ds.GetRasterBand(1).WriteArray(grid.astype("float32")); ds.FlushCache(); ds = None

           dict(sublocation="PW1001.wqual", date_time=datetime(2013, 5, 2, 10, 10), parametername="cond", value="41,0")]
    writers.write_fieldlogger(out / "fieldlogger/fieldlogger_tutorial.csv", obs)


def main(argv: list[str] | None = None) -> int:
    ap = argparse.ArgumentParser(description=__doc__)
    ap.add_argument("--src", type=Path, default=Path("tutorial_data/source"))
    ap.add_argument("--out", type=Path, default=Path("tutorial_data/build"))
    ns = ap.parse_args(argv)
    app = QgsApplication([], False); app.initQgis()
    out = ns.out.resolve()
    if out.exists():
        shutil.rmtree(out)
    out.mkdir(parents=True)
    counts = writers.align_csvs(ns.src, out / "csv", transform_factory())
    dbpath = out / "tutorial.sqlite"
    settings = create_db(dbpath)
    csv_rows = {name: read_csv(out / "csv" / name) for name in counts}
    order = ["obs_points.csv", "obs_lines.csv", "stratigraphy.csv", "w_levels.csv", "w_flow.csv", "w_qual_field.csv",
             "w_qual_lab.csv", "meteo.csv", "seismic_data.csv", "vlf_data.csv"]
    imported = {}
    for name in order:
        table = name[:-4]
        kwargs = {"source_srid": EPSG} if table == "obs_lines" else {}
        imported[table] = import_table(settings, table, csv_rows[name], **kwargs)
    series = split_logger(ns.src / "w_levels_logger.csv")
    add_logger_series(dbpath, series)
    imported["screen"] = add_screens(dbpath)
    add_tem_and_image(dbpath, out)
    add_dem(dbpath, out)
    write_native_files(out, series, csv_rows)
    print(json.dumps({"aligned": counts, "imported": imported}, indent=2))
    app.exitQgis()
    return 0


if __name__ == "__main__":
    sys.exit(main())
```
